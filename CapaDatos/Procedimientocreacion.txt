CREATE PROCEDURE [dbo].[actualizar_cliente]
    @id_cliente INT,              -- Identificador único del cliente a actualizar (obligatorio).
    @nombre VARCHAR(100) = NULL,  -- Nuevo nombre del cliente (opcional).
    @apellido VARCHAR(100) = NULL,  -- Nuevo apellido del cliente (opcional).
    @telefono VARCHAR(15) = NULL,  -- Nuevo número de teléfono (opcional).
    @direccion VARCHAR(255) = NULL,  -- Nueva dirección.
    @correo VARCHAR(150) = NULL  -- Nuevo correo electrónico (opcional).
AS
BEGIN
    SET NOCOUNT ON;  -- Evita que SQL Server envíe mensajes sobre filas afectadas, optimizando el rendimiento.

    -- Actualiza los datos del cliente solo si se proporcionan valores nuevos.
    -- ISNULL(@campo, valor_actual) mantiene el valor original si el parámetro es NULL.
    UPDATE cliente 
    SET nombre = ISNULL(@nombre, nombre),         -- Si @nombre es NULL, conserva el valor actual.
        apellido = ISNULL(@apellido, apellido),   -- Si @apellido es NULL, conserva el valor actual.
        telefono = ISNULL(@telefono, telefono),   -- Si @telefono es NULL, conserva el valor actual.
        direccion = ISNULL(@direccion, direccion), -- Si @direccion es NULL, conserva el valor actual.
        correo = ISNULL(@correo, correo)          -- Si @correo es NULL, conserva el valor actual.
    WHERE id_cliente = @id_cliente;  -- Se actualiza solo el cliente con el ID especificado.
END;

GO

CREATE PROCEDURE actualizar_empleado
    @id_empleado INT,                           -- Parámetro de entrada para identificar al empleado que se va a actualizar
    @nombre NVARCHAR(50),                       -- Parámetro de entrada para el nuevo nombre del empleado
    @apellido NVARCHAR(50),
	@cedula NVARCHAR(13),						-- Parámetro de entrada para el nuevo apellido del empleado
    @telefono NVARCHAR (13),                    -- Parámetro de entrada para el nuevo teléfono del empleado
    @direccion NVARCHAR(100),
	@correo_electronico_empleado VARCHAR(150),                      -- Parámetro de entrada para la nueva dirección del empleado
    @status NVARCHAR(20),                       -- Parámetro de entrada para el nuevo estado del empleado
    @acceso NVARCHAR(20)     -- Parámetro de entrada para el nuevo tipo de acceso del empleado
AS
BEGIN
    -- Actualiza el registro existente en la tabla Empleados que coincide con el id_empleado proporcionado
    UPDATE empleado
    SET nombre = @nombre,                       -- Establece el nuevo nombre
        apellido = @apellido,
		cedula = @cedula,						-- Establece el nuevo apellido
        telefono = @telefono,                   -- Establece el nuevo teléfono
        direccion = @direccion,
		correo_electronico_empleado = @correo_electronico_empleado,                        -- Establece la nueva dirección
        status = @status,                       -- Establece el nuevo estado
        acceso = @acceso                       -- Establece el nuevo tipo de acceso
    WHERE id_empleado = @id_empleado;          -- Condición para identificar el registro a actualizar
END

GO

-- Procedimiento para consultar clientes por ID, nombre o apellido
CREATE PROCEDURE [dbo].[consultar_cliente]
    @pvbusqueda VARCHAR(100)  -- Parámetro de búsqueda (puede ser ID, nombre o apellido)
AS
BEGIN
    SET NOCOUNT ON;  -- Evita mensajes innecesarios de conteo

    IF TRY_CONVERT(INT, @pvbusqueda) IS NOT NULL
    BEGIN
        -- Buscar cliente por ID si el parámetro es un número válido
        SELECT * FROM cliente WHERE id_cliente = CONVERT(INT, @pvbusqueda);
    END
    ELSE
    BEGIN
        -- Buscar cliente por nombre o apellido (búsqueda parcial)
        SELECT * FROM cliente 
        WHERE nombre LIKE '%' + @pvbusqueda + '%' 
           OR apellido LIKE '%' + @pvbusqueda + '%';
    END
END;

GO

CREATE PROCEDURE [dbo].[consultar_factura_producto]
    @pvbusqueda varchar(max) = NULL
AS
BEGIN
    IF TRY_PARSE ( @pvbusqueda AS int ) iS NOT NULL
    BEGIN
		 -- Consulta por id factura
        SELECT fp.id_factura AS factura_id, fp.cantidad AS cantidad,
		pv.precio_venta AS precio_producto, p.nombre AS nombre_producto, (pv.precio_venta * fp.cantidad) AS total_parcial
        FROM factura_producto AS fp
		JOIN inventario AS i ON i.id_inventario = fp.id_inventario
		JOIN proveedor_producto AS pv ON i.id_proveedor_producto = pv.id_proveedor_producto
		JOIN producto AS p ON p.id_producto = pv.id_producto
        WHERE id_factura = @pvbusqueda
    END
   
END

CREATE PROCEDURE [dbo].[consultar_inventario]
	@pvbusqueda varchar(100)
AS
SET NOCOUNT ON;

IF (ISNUMERIC(@pvbusqueda) >= 1)
BEGIN
	SELECT i.*, pr.nombre AS producto_nombre, pv.nombre AS proveedor_nombre
	FROM inventario AS i
	JOIN proveedor_producto AS p ON i.id_proveedor_producto = p.id_proveedor_producto
	JOIN proveedor AS pv ON pv.id_proveedor = p.id_proveedor
	JOIN producto AS pr ON p.id_producto = pr.id_producto
	WHERE id_inventario = @pvbusqueda 
	ORDER BY id_inventario
END
ELSE
BEGIN
	SELECT i.*, pr.nombre AS producto_nombre, pv.nombre AS proveedor_nombre
	FROM inventario AS i
	JOIN proveedor_producto AS p ON i.id_proveedor_producto = p.id_proveedor_producto
	JOIN producto AS pr ON p.id_producto = pr.id_producto
	JOIN proveedor AS pv ON pv.id_proveedor = p.id_proveedor
	WHERE pv.nombre LIKE '%' + @pvbusqueda + '%' OR pr.nombre LIKE '%' + @pvbusqueda + '%'
	ORDER BY i.id_inventario
END

GO



CREATE PROCEDURE [dbo].[consultar_movimiento_inventario]
	@pvbusqueda varchar(100)
AS
SET NOCOUNT ON;

IF (ISNUMERIC(@pvbusqueda) >= 1)
BEGIN
	SELECT mvi.id_movimiento_inventario AS id_movimiento,
		   mvi.id_inventario AS id_inventario,
		   p.id_proveedor_producto AS id_producto,
		   mvi.tipo AS tipo_movimiento,
		   mvi.cantidad AS cantidad_movida,
		   mvi.fecha AS fecha_de_operacion,
		   i.ubicacion AS ubicacion_producto,

	JOIN inventario AS i ON i.id_inventario = mvi.id_inventario
	JOIN proveedor_producto AS p ON i.id_proveedor_producto = p.id_proveedor_producto
	JOIN proveedor AS pv ON pv.id_proveedor = p.id_proveedor
	JOIN producto AS pr ON p.id_producto = pr.id_producto
	WHERE mvi.id_movimiento_inventario = @pvbusqueda 
	ORDER BY mvi.fecha ASC
END
ELSE
BEGIN
	SELECT mvi.id_movimiento_inventario AS id_movimiento,
		   mvi.id_inventario AS id_inventario,
		   p.id_proveedor_producto AS id_producto,
		   mvi.tipo AS tipo_movimiento,
		   mvi.cantidad AS cantidad_movida,
		   mvi.fecha AS fecha_de_operacion,
		   i.ubicacion,
		   pv.nombre AS proveedor_nombre,
		   pr.nombre AS producto_nombre

	JOIN inventario AS i ON i.id_inventario = mvi.id_inventario
	JOIN proveedor_producto AS p ON i.id_proveedor_producto = p.id_proveedor_producto
	JOIN proveedor AS pv ON pv.id_proveedor = p.id_proveedor
	JOIN producto AS pr ON p.id_producto = pr.id_producto

    END
END

-- Procedimiento para insertar un cliente
CREATE PROCEDURE [dbo].[insertar_cliente]
    @id_cliente INT = NULL OUTPUT,   -- Parámetro de salida para obtener el ID generado
    @nombre VARCHAR(100) = 'Cliente de contado',  -- Nombre por defecto
    @apellido VARCHAR(100) = 'Contado',           -- Apellido por defecto
    @telefono VARCHAR(15) = NULL,                 -- Teléfono opcional
    @direccion VARCHAR(255) = NULL,               -- Dirección opcional
    @correo VARCHAR(150) = NULL                   -- Correo opcional
AS
BEGIN
    SET NOCOUNT ON;  -- Evita mensajes innecesarios de conteo de filas afectadas

    -- Inserta un nuevo cliente en la tabla clientes
    INSERT INTO cliente (nombre, apellido, telefono, direccion, correo)

    -- Obtiene el último ID insertado y lo almacena en @id_cliente
    SET @id_cliente = SCOPE_IDENTITY();
END;

GO

CREATE PROCEDURE [dbo].[Consultar_movimiento_inventario_reportes]
	@pvbusqueda INT,  -- Parámetro para determinar la consulta a ejecutar
	@fechainicion date = null,
	@fechafin date = null,
	@id_inventario_producto int = null
AS
BEGIN
    -- Consultando el total movido por fecha
    IF @pvbusqueda = 0
    BEGIN
        SELECT SUM(m.cantidad) AS 'cantidad', CAST(m.fecha AS DATE) AS 'fecha'  -- Convertir fecha a tipo DATE
        FROM movimiento_inventario AS m
        GROUP BY CAST(m.fecha AS DATE)  -- Agrupar solo por la fecha (sin la hora)
        ORDER BY CAST(m.fecha AS DATE) DESC  -- Ordenar solo por la fecha (sin la hora)
    END

    -- Consultando el total movido por tipo de movimiento
    IF @pvbusqueda = 1
    BEGIN
        SELECT SUM(m.cantidad) AS 'cantidad', m.tipo AS 'tipo'
        FROM movimiento_inventario AS m
        GROUP BY m.tipo
        ORDER BY m.tipo ASC
    END

    -- Consultando el total movido por tipo de movimiento y fecha
    IF @pvbusqueda = 2
    BEGIN
        SELECT SUM(m.cantidad) AS 'cantidad', m.tipo AS 'tipo', CAST(m.fecha AS DATE) AS 'fecha'  -- Convertir fecha a tipo DATE
        FROM movimiento_inventario AS m
        GROUP BY m.tipo, CAST(m.fecha AS DATE)  -- Agrupar solo por la fecha (sin la hora)
        ORDER BY m.tipo ASC, CAST(m.fecha AS DATE) DESC  -- Ordenar por tipo y fecha (sin la hora)
    END

	-- Consultando fechas minimas y maximas
	IF @pvbusqueda = 3
	BEGIN
		SELECT CAST(MIN(fecha) AS DATE) AS "fechamin", CAST(MAX(fecha) AS DATE) AS "fechamax"
		FROM movimiento_inventario
	END
	-- Consultando listado movimientos por fecha
	IF @pvbusqueda = 4
	BEGIN
		SELECT mvi.id_movimiento_inventario AS id_movimiento,
		   mvi.id_inventario AS id_inventario,
		   mvi.tipo AS tipo_movimiento,
		   mvi.cantidad AS cantidad_movida,
		   mvi.fecha AS fecha_de_operacion,
		   i.ubicacion AS ubicacion_producto,
		   pv.nombre AS nombre_proveedor,
		   pr.nombre AS nombre_producto,
		   i.id_proveedor_producto AS id_producto
		FROM movimiento_inventario AS mvi
	JOIN inventario AS i ON i.id_inventario = mvi.id_inventario
	JOIN proveedor_producto AS p ON i.id_proveedor_producto = p.id_proveedor_producto
	JOIN proveedor AS pv ON pv.id_proveedor = p.id_proveedor
	JOIN producto AS pr ON p.id_producto = pr.id_producto
	WHERE CAST(mvi.fecha AS DATE) BETWEEN @fechainicion AND @fechafin
	ORDER BY mvi.fecha ASC
	END
	-- Consultando cantidades movidas agrupados por fechas usando BETWEEN
	IF @pvbusqueda = 5
	BEGIN
		SELECT SUM(m.cantidad) AS 'cantidad', CAST(m.fecha AS DATE) AS 'fecha'  -- Convertir fecha a tipo DATE
        FROM movimiento_inventario AS m
		WHERE CAST(m.fecha AS DATE) BETWEEN @fechainicion AND @fechafin
        GROUP BY CAST(m.fecha AS DATE)  -- Agrupar solo por la fecha (sin la hora)
        ORDER BY CAST(m.fecha AS DATE) DESC  -- Ordenar solo por la fecha (sin la hora)
	END

	 -- Consultando el total movido por tipo de movimiento y fecha
    IF @pvbusqueda = 6
    BEGIN
        SELECT SUM(m.cantidad) AS 'cantidad', m.tipo AS 'tipo'
        FROM movimiento_inventario AS m
		WHERE CAST(m.fecha AS DATE) BETWEEN @fechainicion AND @fechafin
        GROUP BY m.tipo
        ORDER BY m.tipo ASC
    END
	-- Consultando los movimientos de un producto en especifico
	IF @pvbusqueda = 7
	BEGIN
		SELECT mvi.id_movimiento_inventario AS id_movimiento,
		   mvi.id_inventario AS id_inventario,
		   mvi.tipo AS tipo_movimiento,
		   mvi.cantidad AS cantidad_movida,
		   mvi.fecha AS fecha_de_operacion,
		   i.ubicacion AS ubicacion_producto,
		   pv.nombre AS nombre_proveedor,
		   pr.nombre AS nombre_producto,
		   p.id_proveedor_producto AS id_producto
		FROM movimiento_inventario AS mvi
	JOIN inventario AS i ON i.id_inventario = mvi.id_inventario
	JOIN proveedor_producto AS p ON i.id_proveedor_producto = p.id_proveedor_producto
	JOIN proveedor AS pv ON pv.id_proveedor = p.id_proveedor
	JOIN producto AS pr ON p.id_producto = pr.id_producto
	WHERE p.id_proveedor_producto = @id_inventario_producto
	ORDER BY mvi.fecha ASC
	END

		-- Consultando los movimientos de un producto en especifico por fechas
	IF @pvbusqueda = 8
	BEGIN
		SELECT mvi.id_movimiento_inventario AS id_movimiento,
		   mvi.id_inventario AS id_inventario,
		   mvi.tipo AS tipo_movimiento,
		   mvi.cantidad AS cantidad_movida,
		   mvi.fecha AS fecha_de_operacion,
		   i.ubicacion AS ubicacion_producto,
		   pv.nombre AS nombre_proveedor,
		   pr.nombre AS nombre_producto,
		   p.id_proveedor_producto AS id_producto
		FROM movimiento_inventario AS mvi
	JOIN inventario AS i ON i.id_inventario = mvi.id_inventario
	JOIN proveedor_producto AS p ON i.id_proveedor_producto = p.id_proveedor_producto
	JOIN proveedor AS pv ON pv.id_proveedor = p.id_proveedor
	JOIN producto AS pr ON p.id_producto = pr.id_producto
	WHERE p.id_proveedor_producto = @id_inventario_producto AND (CAST(mvi.fecha AS DATE ) BETWEEN @fechainicion AND @fechafin)
	ORDER BY mvi.fecha ASC
	END

	-- Consultar Movimientos del producto por fecha
	IF @pvbusqueda = 9
	BEGIN
		SELECT SUM(mvi.cantidad) AS 'cantidad', CAST(mvi.fecha AS DATE) AS 'fecha'  -- Convertir fecha a tipo DATE
        FROM movimiento_inventario AS mvi
		JOIN inventario AS i ON i.id_inventario = mvi.id_inventario
		JOIN proveedor_producto AS p ON p.id_proveedor_producto = @id_inventario_producto
		WHERE CAST(mvi.fecha AS DATE) BETWEEN @fechainicion AND @fechafin AND i.id_proveedor_producto = @id_inventario_producto
        GROUP BY CAST(mvi.fecha AS DATE)  -- Agrupar solo por la fecha (sin la hora)
        ORDER BY CAST(mvi.fecha AS DATE) DESC  -- Ordenar solo por la fecha (sin la hora)
	END

	-- Consultando el total movido por tipo de fecha
	    IF @pvbusqueda = 10
    BEGIN
        SELECT SUM(m.cantidad) AS 'cantidad', m.tipo AS 'tipo'
        FROM movimiento_inventario AS m
		JOIN inventario AS i ON i.id_inventario = m.id_inventario
		JOIN proveedor_producto AS p ON p.id_proveedor_producto = @id_inventario_producto
		WHERE i.id_proveedor_producto = @id_inventario_producto
        GROUP BY m.tipo
        ORDER BY m.tipo ASC
    END

	-- Consultando cada tipo de movimiento inventario por fecha y producto
		IF @pvbusqueda = 11
	BEGIN
		SELECT 
		CAST(m.fecha AS DATE) AS fecha,  -- Agrupar por día
		m.tipo,                           -- Tipo de movimiento (Entrada, Ajuste, Salida)
		SUM(m.cantidad) AS cantidad
		FROM movimiento_inventario AS m
		JOIN inventario AS i ON i.id_inventario = m.id_inventario
		JOIN proveedor_producto AS p ON p.id_proveedor_producto = @id_inventario_producto
		WHERE CAST(m.fecha AS DATE) BETWEEN @fechainicion AND @fechaFin AND i.id_proveedor_producto = @id_inventario_producto
		GROUP BY CAST(m.fecha AS DATE), m.tipo
		ORDER BY fecha ASC, m.tipo;
	END

	-- Consultando cada tipo de movimiento inventario por fecha
		IF @pvbusqueda = 12
	BEGIN
		SELECT 
		CAST(m.fecha AS DATE) AS fecha,  -- Agrupar por día
		m.tipo,                           -- Tipo de movimiento (Entrada, Ajuste, Salida)
		SUM(m.cantidad) AS cantidad
		FROM movimiento_inventario AS m
		JOIN inventario AS i ON i.id_inventario = m.id_inventario
		JOIN proveedor_producto AS p ON p.id_proveedor_producto = i.id_proveedor_producto
		WHERE CAST(m.fecha AS DATE) BETWEEN @fechainicion AND @fechaFin
		GROUP BY CAST(m.fecha as DATE), m.tipo
		ORDER BY fecha ASC, m.tipo;
	END

END
